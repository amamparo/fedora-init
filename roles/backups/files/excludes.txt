# fedora-init (backups role) — what restic skips inside $HOME. Rebuildable
# trees and caches only: a restore should bring back work, not re-download
# state the roles or package managers recreate anyway.
#
# Pattern semantics (restic): a bare name matches ANY path component with
# that name, anywhere; ** spans directories. Comments must sit on their own
# lines — restic reads trailing text as part of the pattern.

# Caches everywhere (CACHEDIR.TAG-marked ones are also caught by the
# script's --exclude-caches)
.cache

# Dependency/build output inside checkouts — restored code re-fetches these
node_modules
.venv
__pycache__

# Btrfs snapshot mounts, should any ever appear under $HOME — backing up
# snapshots-of-home into a backup-of-home would recurse badly
.snapshots

# Podman image/layer storage — rebuildable from registries
**/.local/share/containers

# Steam game installs — tens of GB, all re-downloadable
**/.local/share/Steam

# Self-updating app bundles the roles reinstall from scratch anyway
**/.local/share/claude
**/.local/share/JetBrains/Toolbox

# Runtimes mise re-creates from per-project pins
**/.local/share/mise

# Trash
**/.local/share/Trash

**/Music/REAPER/Peaks
